cmake_minimum_required(VERSION 3.18)
project(triangle_counting LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenMP REQUIRED COMPONENTS CXX)
add_library(tc_preprocess STATIC src/common/preprocess.cpp)
target_link_libraries(tc_preprocess PUBLIC OpenMP::OpenMP_CXX)
add_executable(tc_cpu_serial src/cpu/serial.cpp)
target_link_libraries(tc_cpu_serial PRIVATE tc_preprocess)
option(TC_ENABLE_CUDA "Build serial, edge, warp, split and experimental query CUDA implementations" ON)
if(TC_ENABLE_CUDA)
    enable_language(CUDA)
    add_executable(tc_gpu_serial src/cuda/serial.cu)
    add_executable(tc_gpu_edge_baseline src/cuda/edge_baseline.cu)
    target_link_libraries(tc_gpu_edge_baseline PRIVATE tc_preprocess)
    # A shared runner keeps preprocessing and timing boundaries identical.
    add_executable(tc_gpu_warp src/cuda/edge_baseline.cu)
    target_compile_definitions(tc_gpu_warp PRIVATE TC_EDGE_MODE=1)
    target_link_libraries(tc_gpu_warp PRIVATE tc_preprocess)
    add_executable(tc_gpu_split src/cuda/edge_baseline.cu)
    target_compile_definitions(tc_gpu_split PRIVATE TC_EDGE_MODE=2)
    target_link_libraries(tc_gpu_split PRIVATE tc_preprocess)
    add_executable(tc_gpu_query src/cuda/edge_baseline.cu)
    target_compile_definitions(tc_gpu_query PRIVATE TC_EDGE_MODE=3)
    target_link_libraries(tc_gpu_query PRIVATE tc_preprocess)
    add_executable(tc_gpu_query_v2 src/cuda/edge_baseline.cu)
    target_compile_definitions(tc_gpu_query_v2 PRIVATE TC_EDGE_MODE=4)
    target_link_libraries(tc_gpu_query_v2 PRIVATE tc_preprocess)
    add_executable(tc_gpu_query_global src/cuda/edge_baseline.cu)
    target_compile_definitions(tc_gpu_query_global PRIVATE TC_EDGE_MODE=5)
    target_link_libraries(tc_gpu_query_global PRIVATE tc_preprocess)
    add_executable(tc_schedule_calibrate src/cuda/schedule_runner.cu)
    target_compile_definitions(tc_schedule_calibrate PRIVATE TC_SCHEDULE_CALIBRATE=1)
    target_link_libraries(tc_schedule_calibrate PRIVATE tc_preprocess)
    add_executable(tc_schedule_auto src/cuda/schedule_runner.cu)
    target_link_libraries(tc_schedule_auto PRIVATE tc_preprocess)
    set_target_properties(tc_schedule_calibrate tc_schedule_auto PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
    set_target_properties(tc_gpu_serial tc_gpu_edge_baseline tc_gpu_warp tc_gpu_split tc_gpu_query tc_gpu_query_v2 tc_gpu_query_global PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
endif()

include(CTest)

    add_executable(tc_schedule_policy_checks tests/test_schedule_policy.cpp)
    add_test(NAME schedule_policy_host COMMAND tc_schedule_policy_checks)
    add_executable(tc_global_query_checks tests/test_global_queries.cpp)
    add_test(NAME global_queries_host COMMAND tc_global_query_checks)
    add_executable(tc_query_v2_checks tests/test_query_v2.cpp)
    add_test(NAME query_v2_host COMMAND tc_query_v2_checks)
    add_executable(tc_query_policy_checks tests/test_query_policy.cpp)
    add_test(NAME query_policy_host COMMAND tc_query_policy_checks)
    add_executable(tc_edge_binary_checks tests/test_edge_binary.cpp)
    add_executable(tc_warp_task_checks tests/test_warp_tasks.cpp)
    add_test(NAME warp_tasks_host COMMAND tc_warp_task_checks)
    add_executable(tc_preprocess_equivalence tests/preprocess_equivalence.cpp)
    target_link_libraries(tc_preprocess_equivalence PRIVATE tc_preprocess)
    add_test(NAME preprocess_equivalence COMMAND tc_preprocess_equivalence "${CMAKE_CURRENT_SOURCE_DIR}/tests/triangle.txt")
    add_test(NAME edge_binary_host COMMAND tc_edge_binary_checks "${CMAKE_CURRENT_SOURCE_DIR}/tests/triangle.txt")
endif()
