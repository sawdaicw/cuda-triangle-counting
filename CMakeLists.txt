cmake_minimum_required(VERSION 3.18)
project(triangle_counting LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(tc_cpu_serial src/cpu/serial.cpp)
option(TC_ENABLE_CUDA "Build the single-thread CUDA implementation" ON)
if(TC_ENABLE_CUDA)
    enable_language(CUDA)
    add_executable(tc_gpu_serial src/cuda/serial.cu)
    set_target_properties(tc_gpu_serial PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
endif()
