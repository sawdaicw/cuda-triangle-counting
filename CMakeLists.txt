cmake_minimum_required(VERSION 3.18)
project(triangle_counting LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenMP REQUIRED COMPONENTS CXX)
add_library(tc_preprocess STATIC src/common/preprocess.cpp)
target_link_libraries(tc_preprocess PUBLIC OpenMP::OpenMP_CXX)
add_executable(tc_cpu_serial src/cpu/serial.cpp)
target_link_libraries(tc_cpu_serial PRIVATE tc_preprocess)
option(TC_ENABLE_CUDA "Build the serial and edge-parallel CUDA implementations" ON)
if(TC_ENABLE_CUDA)
    enable_language(CUDA)
    add_executable(tc_gpu_serial src/cuda/serial.cu)
    add_executable(tc_gpu_edge_baseline src/cuda/edge_baseline.cu)
    target_link_libraries(tc_gpu_edge_baseline PRIVATE tc_preprocess)
    set_target_properties(tc_gpu_serial tc_gpu_edge_baseline PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(tc_edge_binary_checks tests/test_edge_binary.cpp)
    add_executable(tc_preprocess_equivalence tests/preprocess_equivalence.cpp)
    target_link_libraries(tc_preprocess_equivalence PRIVATE tc_preprocess)
    add_test(NAME preprocess_equivalence COMMAND tc_preprocess_equivalence "${CMAKE_CURRENT_SOURCE_DIR}/tests/triangle.txt")
    add_test(NAME edge_binary_host COMMAND tc_edge_binary_checks "${CMAKE_CURRENT_SOURCE_DIR}/tests/triangle.txt")
endif()
